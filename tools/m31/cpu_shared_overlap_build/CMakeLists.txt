cmake_minimum_required(VERSION 3.22)
project(m31_shared_overlap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(M31_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(M31_BASE "${M31_ROOT}/build-m31-cuda")
set(M31_VENDOR "${M31_ROOT}/build-m31-cpu/_deps/strata_llamacpp-src")
find_package(CUDAToolkit REQUIRED)
add_library(m31_overlap_engine STATIC ../m31_engine.cpp ../m31_gpu_experts.cpp)
target_compile_definitions(m31_overlap_engine PRIVATE M31_ENGINE_CUDA=1)
option(M31_SHARED_OVERLAP_TESTING "Compile failure hook for the isolated proof" ON)
if(M31_SHARED_OVERLAP_TESTING)
  target_compile_definitions(m31_overlap_engine PRIVATE M31_SHARED_OVERLAP_TESTING=1)
endif()
target_include_directories(m31_overlap_engine PUBLIC "${M31_ROOT}/include" "${M31_ROOT}/tools/m31" "${M31_VENDOR}/ggml/include")
target_link_libraries(m31_overlap_engine PUBLIC "${M31_BASE}/ggml/src/libggml.a" "${M31_BASE}/ggml/src/libggml-cpu.a" "${M31_BASE}/ggml/src/ggml-cuda/libggml-cuda.a" "${M31_BASE}/ggml/src/libggml-base.a" CUDA::cudart_static CUDA::cublas_static CUDA::cublasLt_static CUDA::culibos CUDA::cuda_driver m dl rt pthread)
option(M31_SHARED_OVERLAP_NVTX "Compile optional proof-only CPU ranges" ON)
if(M31_SHARED_OVERLAP_NVTX)
  target_compile_definitions(m31_overlap_engine PRIVATE M31_SHARED_OVERLAP_NVTX=1)
  target_include_directories(m31_overlap_engine PRIVATE /opt/nvidia/nsight-systems/2025.5.2/target-linux-x64/nvtx/include)
endif()
add_executable(cpu_shared_overlap_check ../cpu_shared_overlap_check.cpp)
target_compile_definitions(cpu_shared_overlap_check PRIVATE M31_HAVE_CUDA=1)
target_link_libraries(cpu_shared_overlap_check PRIVATE m31_overlap_engine)
add_executable(m31_generate ../m31_generate.cpp)
target_compile_definitions(m31_generate PRIVATE M31_HAVE_CUDA=1)
target_link_libraries(m31_generate PRIVATE m31_overlap_engine)
add_executable(m31_serve ../m31_serve.cpp)
target_compile_definitions(m31_serve PRIVATE M31_HAVE_CUDA=1)
target_include_directories(m31_serve PRIVATE "${M31_VENDOR}/vendor/nlohmann")
target_link_libraries(m31_serve PRIVATE m31_overlap_engine)
add_executable(m31_inspect ../m31_inspect.cpp)
target_include_directories(m31_inspect PRIVATE "${M31_ROOT}/include" "${M31_ROOT}/tools/m31")
