cmake_minimum_required(VERSION 3.22)
project(m31_cpu_decode LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(M31_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(M31_BASE "${M31_ROOT}/build-m31-cuda" CACHE PATH "Completed CUDA vendor build")
set(M31_VENDOR "${M31_ROOT}/build-m31-cpu/_deps/strata_llamacpp-src")
find_package(CUDAToolkit REQUIRED)
add_library(m31_engine STATIC ../m31_engine.cpp ../m31_gpu_experts.cpp)
target_compile_definitions(m31_engine PRIVATE M31_ENGINE_CUDA=1)
target_include_directories(m31_engine PUBLIC "${M31_ROOT}/include" "${M31_ROOT}/tools/m31" "${M31_VENDOR}/ggml/include")
target_link_libraries(m31_engine PUBLIC "${M31_BASE}/ggml/src/libggml.a" "${M31_BASE}/ggml/src/libggml-cpu.a" "${M31_BASE}/ggml/src/ggml-cuda/libggml-cuda.a" "${M31_BASE}/ggml/src/libggml-base.a" CUDA::cudart_static CUDA::cublas_static CUDA::cublasLt_static CUDA::culibos CUDA::cuda_driver m dl rt pthread)
foreach(program generate serve)
  add_executable(m31_${program} ../m31_${program}.cpp)
  target_compile_definitions(m31_${program} PRIVATE M31_HAVE_CUDA=1)
  target_link_libraries(m31_${program} PRIVATE m31_engine)
endforeach()
target_include_directories(m31_serve PRIVATE "${M31_VENDOR}/vendor/nlohmann")
add_executable(m31_inspect ../m31_inspect.cpp)
target_include_directories(m31_inspect PRIVATE "${M31_ROOT}/include" "${M31_ROOT}/tools/m31")
