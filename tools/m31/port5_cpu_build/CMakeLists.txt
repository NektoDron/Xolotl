cmake_minimum_required(VERSION 3.22)
project(m31_port5_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(M31_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(M31_VENDOR "${M31_ROOT}/build-m31-cpu/_deps/strata_llamacpp-src")
add_executable(port5_cpu_probe ../port5_cpu_probe.cpp ../cpu_gpu_host_executor.cpp)
target_compile_options(port5_cpu_probe PRIVATE -O3 -frounding-math -fno-fast-math)
target_include_directories(port5_cpu_probe PRIVATE "${M31_ROOT}/include" "${M31_ROOT}/tools/m31" "${M31_VENDOR}/ggml/include")
target_link_libraries(port5_cpu_probe PRIVATE "${M31_ROOT}/build-m31-cpu/ggml/src/libggml.a" "${M31_ROOT}/build-m31-cpu/ggml/src/libggml-cpu.a" "${M31_ROOT}/build-m31-cpu/ggml/src/libggml-base.a" pthread m dl)
add_executable(port5_cpu_affinity_check ../port5_cpu_affinity_check.cpp ../cpu_gpu_host_executor.cpp)
target_include_directories(port5_cpu_affinity_check PRIVATE "${M31_ROOT}/tools/m31" "${M31_VENDOR}/ggml/include")
target_link_libraries(port5_cpu_affinity_check PRIVATE "${M31_ROOT}/build-m31-cpu/ggml/src/libggml-base.a" pthread m dl)
add_executable(port5_cpu_pack_check ../port5_cpu_pack_check.cpp ../cpu_gpu_host_executor.cpp)
target_include_directories(port5_cpu_pack_check PRIVATE "${M31_ROOT}/tools/m31")
target_link_libraries(port5_cpu_pack_check PRIVATE pthread m)
