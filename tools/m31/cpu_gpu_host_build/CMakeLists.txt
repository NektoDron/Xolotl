cmake_minimum_required(VERSION 3.22)
project(m31_gpu_host_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(M31_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(M31_VENDOR "${M31_ROOT}/build-m31-cpu/_deps/strata_llamacpp-src")
find_package(Threads REQUIRED)
add_library(cpu_gpu_host STATIC ../cpu_gpu_host_executor.cpp ../cpu_gpu_host_ops.cpp)
target_compile_options(cpu_gpu_host PRIVATE -frounding-math -fno-fast-math)
target_include_directories(cpu_gpu_host PUBLIC "${M31_ROOT}/tools/m31" "${M31_VENDOR}/ggml/include")
target_link_libraries(cpu_gpu_host PUBLIC "${M31_ROOT}/build-m31-cpu/ggml/src/libggml.a" "${M31_ROOT}/build-m31-cpu/ggml/src/libggml-cpu.a" "${M31_ROOT}/build-m31-cpu/ggml/src/libggml-base.a" Threads::Threads m dl)
add_executable(cpu_gpu_host_test ../cpu_gpu_host_test.cpp)
target_compile_options(cpu_gpu_host_test PRIVATE -frounding-math -fno-fast-math)
target_link_libraries(cpu_gpu_host_test PRIVATE cpu_gpu_host)
add_library(cpu_gpu_host_cpp17 OBJECT ../cpu_gpu_host_cpp17.cpp)
set_target_properties(cpu_gpu_host_cpp17 PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_include_directories(cpu_gpu_host_cpp17 PRIVATE "${M31_ROOT}/tools/m31" "${M31_VENDOR}/ggml/include")
